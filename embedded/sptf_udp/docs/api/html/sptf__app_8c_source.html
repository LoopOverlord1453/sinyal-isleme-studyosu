79static uint64_t board_time_ns(void *ctx)
80{
81 static uint32_t last = 0u;
82 static uint64_t high = 0u;
83 const uint32_t now = DWT->CYCCNT;
84 const uint64_t hz = (uint64_t)SystemCoreClock;
85 uint64_t cycles;
86 (void)ctx;
87 if (now < last)
88 {
89 high += (uint64_t)1u << 32u;
90 }
91 last = now;
92 cycles = high | (uint64_t)now;
93 return ((cycles / hz) * 1000000000u) + (((cycles % hz) * 1000000000u) / hz);
94}

139int sptf_app_init(void)
140{
141 sptf_link_config_t lc = {0};
142 ip_addr_t pc;
143 sptf_status_t st;
144 dwt_enable();
145 APP_IP_ADDR4(&pc, SPTF_APP_PC_IP);
146 st = sptf_lwip_raw_open(&s_port, &pc, (u16_t)SPTF_UDP_PORT, 0u);
147 if (st == SPTF_OK)
148 {
149 sptf_lwip_raw_transport(&s_port, &lc.transport);
150 lc.clock = board_time_ns;
151 st = sptf_link_init(&s_link, &lc);
152 }
153 if (st == SPTF_OK)
154 {
155 st = sptf_stream_init(&s_stream, &s_link, &s_cfg);
156 }
157 if ((st == SPTF_OK) && (HAL_ADC_Start_DMA(&hadc1, (uint32_t *)s_adc, 2u * APP_HALF) != HAL_OK))
158 {
159 st = SPTF_ERR_IO;
160 }
161 return (int)st;
162}

165static void send_half(uint32_t half)
166{
167 uint16_t *p = &s_adc[half * APP_HALF];
168#if defined(__DCACHE_PRESENT) && (__DCACHE_PRESENT == 1U)
169 SCB_InvalidateDCache_by_Addr(p, (int32_t)(APP_HALF * sizeof(uint16_t)));
170#endif
171 (void)sptf_stream_write(&s_stream, p, SPTF_APP_BLOCK, SPTF_TS_AUTO);
172}

174void sptf_app_process(void)
175{
176 uint8_t ready;
177 uint32_t now_ms;
178 const uint32_t primask = __get_PRIMASK();
179 __disable_irq();
180 ready = s_ready;
181 s_ready = 0u;
182 __set_PRIMASK(primask);
183 if ((ready & 1u) != 0u)
184 {
185 send_half(0u);
186 }
187 if ((ready & 2u) != 0u)
188 {
189 send_half(1u);
190 }
191 if (s_overruns != s_reported)
192 {
193 s_reported = s_overruns;
194 (void)sptf_stream_marker(&s_stream, SPTF_MARKER_AT_NEXT, 0xDEADu, "ADC taşması (ana döngü yavaş)", SPTF_TS_AUTO);
195 }
196 now_ms = HAL_GetTick();
197 if ((now_ms - s_last_poll) >= 100u)
198 {
199 s_last_poll = now_ms;
200 (void)sptf_link_poll(&s_link);
201 }
202}
