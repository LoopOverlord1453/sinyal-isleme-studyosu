36static sptf_status_t map_err(err_t err)
37{
38 sptf_status_t st;
39 switch (err)
40 {
41 case ERR_OK:
42 st = SPTF_OK;
43 break;
44 case ERR_MEM:
45 case ERR_BUF:
46 st = SPTF_ERR_NOMEM;
47 break;
48 case ERR_WOULDBLOCK:
49 case ERR_INPROGRESS:
50 st = SPTF_ERR_BUSY;
51 break;
52 default:
53 st = SPTF_ERR_IO;
54 break;
55 }
56 return st;
57}

60static uint8_t *lwip_acquire(void *ctx, size_t len)
61{
62 sptf_lwip_raw_t *port = (sptf_lwip_raw_t *)ctx;
63 struct pbuf *pb = NULL;
64 if ((port == NULL) || (len == 0u) || (len > 0xFFFFu))
65 {
66 return NULL;
67 }
68 SPTF_LWIP_LOCK();
69 if (port->pending != NULL)
70 {
71 /* Sözleşme ihlali (acquire/send eşleşmedi): sızıntıyı önle. */
72 (void)pbuf_free(port->pending);
73 port->pending = NULL;
74 }
75 pb = pbuf_alloc(PBUF_TRANSPORT, (u16_t)len, PBUF_RAM);
76 port->pending = pb;
77 SPTF_LWIP_UNLOCK();
78 return (pb != NULL) ? (uint8_t *)pb->payload : NULL;
79}

82static sptf_status_t lwip_send(void *ctx, uint8_t *buf, size_t len)
83{
84 sptf_lwip_raw_t *port = (sptf_lwip_raw_t *)ctx;
85 sptf_status_t st = SPTF_OK;
86 struct pbuf *pb;
87 if (port == NULL)
88 {
89 return SPTF_ERR_ARG;
90 }
91 SPTF_LWIP_LOCK();
92 pb = port->pending;
93 port->pending = NULL;
94 if ((pb == NULL) || ((uint8_t *)pb->payload != buf) || (len > (size_t)pb->tot_len))
95 {
96 st = SPTF_ERR_ARG;
97 }
98 else if (len > 0u)
99 {
100 pbuf_realloc(pb, (u16_t)len);
101 st = map_err(udp_sendto(port->pcb, pb, &port->dst_ip, port->dst_port));
102 }
103 else
104 {
105 /* len == 0: yalnız serbest bırak */
106 }
107 if (pb != NULL)
108 {
109 (void)pbuf_free(pb);
110 }
111 SPTF_LWIP_UNLOCK();
112 return st;
113}

115sptf_status_t sptf_lwip_raw_open(sptf_lwip_raw_t *port, const ip_addr_t *dst_ip, u16_t dst_port, u16_t local_port)
116{
117 sptf_status_t st = SPTF_OK;
118 if ((port == NULL) || (dst_ip == NULL))
119 {
120 return SPTF_ERR_ARG;
121 }
122 port->pending = NULL;
123 port->dst_port = dst_port;
124 ip_addr_copy(port->dst_ip, *dst_ip);
125 SPTF_LWIP_LOCK();
126 port->pcb = udp_new();
127 if (port->pcb == NULL)
128 {
129 st = SPTF_ERR_NOMEM;
130 }
131 else if (udp_bind(port->pcb, IP_ANY_TYPE, local_port) != (err_t)ERR_OK)
132 {
133 udp_remove(port->pcb);
134 port->pcb = NULL;
135 st = SPTF_ERR_IO;
136 }
137 else
138 {
139 /* hazır */
140 }
141 SPTF_LWIP_UNLOCK();
142 return st;
143}

159{

161 {
162 SPTF_LWIP_LOCK();
163 if (port->pending != NULL)
164 {
165 (void)pbuf_free(port->pending);

173 SPTF_LWIP_UNLOCK();
